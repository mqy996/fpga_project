cmake_minimum_required(VERSION 3.21)
project(QtControlHost_v6 VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network)

qt_add_executable(QtControlHost_v6 WIN32
    src/main.cpp
    src/protocol.h src/protocol.cpp
    src/controlclient.h src/controlclient.cpp
    src/adcmath.h
    src/csvrecorder.h src/csvrecorder.cpp
    src/dataworker.h src/dataworker.cpp
    src/wavewidget.h src/wavewidget.cpp
    src/mainwindow.h src/mainwindow.cpp
)

target_link_libraries(QtControlHost_v6 PRIVATE
    Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network ws2_32
)
target_include_directories(QtControlHost_v6 PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)

target_compile_definitions(QtControlHost_v6 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)

include(CTest)
if(BUILD_TESTING)
    qt_add_executable(protocol_tests tests/protocol_tests.cpp src/protocol.h src/protocol.cpp)
    target_link_libraries(protocol_tests PRIVATE Qt6::Core Qt6::Network)
    target_include_directories(protocol_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
    add_test(NAME protocol_tests COMMAND protocol_tests)
    qt_add_executable(controlclient_mock_tests
        tests/controlclient_mock_tests.cpp
        src/protocol.h src/protocol.cpp src/controlclient.h src/controlclient.cpp
    )
    target_link_libraries(controlclient_mock_tests PRIVATE Qt6::Core Qt6::Network)
    target_include_directories(controlclient_mock_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
    add_test(NAME controlclient_mock_tests COMMAND controlclient_mock_tests)
    qt_add_executable(monitor_math_tests tests/monitor_math_tests.cpp src/adcmath.h)
    target_link_libraries(monitor_math_tests PRIVATE Qt6::Core Qt6::Network)
    target_include_directories(monitor_math_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
    add_test(NAME monitor_math_tests COMMAND monitor_math_tests)
endif()
